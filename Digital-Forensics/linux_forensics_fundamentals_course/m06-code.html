Create a File Metadata Timeline

ROOT=/mnt/evidence/rootfs

find "$ROOT" -xdev -type f \
  -printf '%T@|mtime|%TY-%Tm-%TdT%TH:%TM:%TS%Tz|%u|%g|%m|%s|%p\n' 2>/dev/null | \
  sort -n > case-file-timeline.tsv

Find Recently Modified Executables and Dotfiles

ROOT=/mnt/evidence/rootfs

find "$ROOT/home" "$ROOT/root" "$ROOT/tmp" "$ROOT/var/tmp" -xdev \
  \( -perm -111 -o -name '.*' \) \
  -type f -mtime -14 -printf '%TY-%Tm-%Td %TH:%TM %m %u:%g %s %p\n' 2>/dev/null | sort

Merge Simple Timeline Rows

from pathlib import Path

rows = []
for line in Path("case-file-timeline.tsv").read_text(errors="replace").splitlines():
    epoch, kind, iso_time, user, group, mode, size, path = line.split("|", 7)
    rows.append((float(epoch), iso_time, kind, user, mode, size, path))

for _, iso_time, kind, user, mode, size, path in sorted(rows)[-25:]:
    print(f"{iso_time} {kind:5} {user:>8} {mode} {size:>9} {path}")
