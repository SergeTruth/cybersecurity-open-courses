Hash a Collected Image Before Analysis

CASE=case-2026-06-26
IMAGE=/mnt/evidence/linux-disk.img

mkdir -p "$CASE"
sha256sum "$IMAGE" | tee "$CASE/evidence-hashes.txt"
stat --printf 'name=%n\nsize=%s\nmodified=%y\n' "$IMAGE" | tee "$CASE/evidence-stat.txt"

Create a Minimal Case Note

CASE=case-2026-06-26

{
  printf 'case_id=%s\n' "$CASE"
  printf 'analyst=%s\n' "$USER"
  printf 'started_utc=%s\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)"
  printf 'scope=%s\n' 'authorized Linux forensic review'
} | tee "$CASE/case-notes.txt"
