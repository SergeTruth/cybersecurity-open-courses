Find Shell-Enabled Local Accounts

ROOT=/mnt/evidence/rootfs

awk -F: '$7 !~ /(false|nologin)$/ {
  printf "user=%s uid=%s gid=%s home=%s shell=%s\n", $1, $3, $4, $6, $7
}' "$ROOT/etc/passwd"

Review Administrative Group Membership

ROOT=/mnt/evidence/rootfs

grep -E '^(sudo|wheel|adm):' "$ROOT/etc/group" 2>/dev/null
find "$ROOT/etc/sudoers" "$ROOT/etc/sudoers.d" -maxdepth 1 -type f -print 2>/dev/null

List SSH Keys and Shell Histories

ROOT=/mnt/evidence/rootfs

find "$ROOT/home" "$ROOT/root" -xdev \( -name authorized_keys -o -name known_hosts -o -name '.*history' \) \
  -type f -printf '%TY-%Tm-%Td %TH:%TM %u:%g %m %p\n' 2>/dev/null

Parse /etc/passwd Into Structured Rows

from pathlib import Path

passwd = Path("/mnt/evidence/rootfs/etc/passwd")
for line in passwd.read_text(encoding="utf-8", errors="replace").splitlines():
    user, _, uid, gid, comment, home, shell = line.split(":", 6)
    if shell.endswith(("nologin", "false")):
        continue
    print({"user": user, "uid": int(uid), "gid": int(gid), "home": home, "shell": shell})
