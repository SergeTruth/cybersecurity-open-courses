Search Rotated Authentication Logs

ROOT=/mnt/evidence/rootfs

zgrep -hE 'Accepted|Failed|sudo|session opened|session closed' \
  "$ROOT"/var/log/auth.log* "$ROOT"/var/log/secure* 2>/dev/null | \
  tee case-auth-events.txt

Read an Offline systemd Journal

ROOT=/mnt/evidence/rootfs

journalctl --directory "$ROOT/var/log/journal" \
  --since '2026-06-01 00:00:00' \
  --output short-iso \
  --no-pager | tee case-journal-events.txt

Count SSH Login Outcomes by Source

import re
from collections import Counter
from pathlib import Path

pattern = re.compile(r"(Accepted|Failed).* from (?P<ip>\S+)")
counts = Counter()

for path in Path("/mnt/evidence/rootfs/var/log").glob("auth.log*"):
    text = path.read_text(encoding="utf-8", errors="replace")
    for match in pattern.finditer(text):
        counts[(match.group(1), match.group("ip"))] += 1

for (result, ip), count in counts.most_common(20):
    print(f"{result:8} {ip:>15} {count}")
