Identify Distribution and Release

ROOT=/mnt/evidence/rootfs

cat "$ROOT/etc/os-release"
cat "$ROOT/etc/issue" 2>/dev/null || true
find "$ROOT/etc" -maxdepth 1 \( -name '*release' -o -name '*version' \) -type f -print

Inventory High-Value Directories

ROOT=/mnt/evidence/rootfs

for path in etc var/log home root tmp opt usr/local; do
  printf '\n[%s]\n' "$ROOT/$path"
  find "$ROOT/$path" -maxdepth 2 -xdev -type f -printf '%TY-%Tm-%Td %TH:%TM %s %p\n' 2>/dev/null | head -50
done

Summarize Mounted Evidence Layout

from pathlib import Path

root = Path("/mnt/evidence/rootfs")
for name in ["etc", "var/log", "home", "root", "tmp", "opt"]:
    path = root / name
    files = sum(1 for item in path.rglob("*") if item.is_file()) if path.exists() else 0
    print(f"{path}: {files} files")
