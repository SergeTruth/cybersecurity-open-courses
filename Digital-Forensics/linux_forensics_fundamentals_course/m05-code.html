Inspect systemd Unit Files in Evidence

ROOT=/mnt/evidence/rootfs

find "$ROOT/etc/systemd/system" "$ROOT/usr/lib/systemd/system" \
  -type f -name '*.service' -printf '%TY-%Tm-%Td %TH:%TM %p\n' 2>/dev/null | sort

grep -RHE '^(ExecStart|User|Group|WantedBy)=' \
  "$ROOT/etc/systemd/system" "$ROOT/usr/lib/systemd/system" 2>/dev/null

Review Scheduled Task Locations

ROOT=/mnt/evidence/rootfs

find "$ROOT/etc/cron.d" "$ROOT/etc/cron.daily" "$ROOT/etc/cron.hourly" \
     "$ROOT/etc/cron.monthly" "$ROOT/etc/cron.weekly" "$ROOT/var/spool/cron" \
  -type f -printf '%TY-%Tm-%Td %TH:%TM %u:%g %m %p\n' 2>/dev/null | sort
