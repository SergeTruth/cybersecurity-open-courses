Extract SSH Login Sources

ROOT=/mnt/evidence/rootfs

zgrep -hE 'Accepted|Failed' "$ROOT"/var/log/auth.log* "$ROOT"/var/log/secure* 2>/dev/null | \
  awk 'match($0, /from [0-9.]+/) { print substr($0, RSTART + 5, RLENGTH - 5) }' | \
  sort | uniq -c | sort -nr

Fingerprint Authorized SSH Keys

ROOT=/mnt/evidence/rootfs

find "$ROOT/home" "$ROOT/root" -path '*/.ssh/authorized_keys' -type f -print0 2>/dev/null | \
  while IFS= read -r -d '' keys; do
    printf '\n[%s]\n' "$keys"
    ssh-keygen -lf "$keys" 2>/dev/null
  done

Review SSH and Firewall Configuration

ROOT=/mnt/evidence/rootfs

grep -RHE '^(Port|ListenAddress|PermitRootLogin|PasswordAuthentication|AllowUsers|AllowGroups)' \
  "$ROOT/etc/ssh/sshd_config" "$ROOT/etc/ssh/sshd_config.d" 2>/dev/null

find "$ROOT/etc" -maxdepth 3 \( -name '*iptables*' -o -name '*nft*' -o -name '*ufw*' \) -type f -print
