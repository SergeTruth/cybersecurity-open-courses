Build a Small Cross-Artifact Timeline

from pathlib import Path

events = []

for path in Path("case").glob("*auth-events.txt"):
    for line in path.read_text(errors="replace").splitlines():
        if "Accepted" in line or "Failed" in line:
            events.append(("auth", line[:15], line))

for line in Path("case-file-timeline.tsv").read_text(errors="replace").splitlines():
    parts = line.split("|", 7)
    if len(parts) == 8 and ("/.ssh/" in parts[7] or "/systemd/" in parts[7]):
        events.append(("file", parts[2], parts[7]))

for kind, when, detail in sorted(events, key=lambda item: item[1]):
    print(f"{when} [{kind}] {detail}")

Quick Read-Only Triage Checklist

ROOT=/mnt/evidence/rootfs

printf '\n== OS ==\n'; cat "$ROOT/etc/os-release" 2>/dev/null
printf '\n== Shell users ==\n'; awk -F: '$7 !~ /(false|nologin)$/ {print $1,$3,$6,$7}' "$ROOT/etc/passwd"
printf '\n== Recent auth events ==\n'; zgrep -hE 'Accepted|Failed|sudo' "$ROOT"/var/log/auth.log* 2>/dev/null | tail -50
printf '\n== Service changes ==\n'; find "$ROOT/etc/systemd/system" -type f -mtime -30 -print 2>/dev/null
printf '\n== SSH keys ==\n'; find "$ROOT/home" "$ROOT/root" -path '*/.ssh/authorized_keys' -type f -print 2>/dev/null
