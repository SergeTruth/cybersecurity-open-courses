Parse an Integer Token with from_chars

#include <charconv>
#include <optional>
#include <string_view>
#include <system_error>

std::optional<unsigned short> parse_port(std::string_view text) {
    unsigned int value = 0;
    const char* first = text.data();
    const char* last = first + text.size();
    auto [ptr, ec] = std::from_chars(first, last, value);

    if (ec != std::errc{} || ptr != last) {
        return std::nullopt;
    }
    if (value < 1 || value > 65535) {
        return std::nullopt;
    }

    return static_cast<unsigned short>(value);
}

Check Size Math Before Allocation

#include <cstddef>
#include <limits>
#include <optional>
#include <vector>

struct Record {
    unsigned int id;
    unsigned int flags;
};

std::optional<std::vector<Record>> allocate_records(std::size_t count) {
    constexpr std::size_t max_records = 100000;
    constexpr std::size_t max_bytes = 8 * 1024 * 1024;

    if (count > max_records) {
        return std::nullopt;
    }
    if (count > std::numeric_limits<std::size_t>::max() / sizeof(Record)) {
        return std::nullopt;
    }
    if (count * sizeof(Record) > max_bytes) {
        return std::nullopt;
    }

    return std::vector<Record>(count);
}
