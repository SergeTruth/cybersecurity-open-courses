Validate Indexes Before Container Access

#include <cstddef>
#include <optional>
#include <string>
#include <vector>

struct Account {
    std::string name;
    bool enabled;
};

std::optional<Account> get_account(const std::vector<Account>& accounts, std::size_t index) {
    if (index >= accounts.size()) {
        return std::nullopt;
    }

    return accounts[index];
}

Reject Input That Would Grow Containers Too Far

#include <cstddef>
#include <optional>
#include <string>
#include <utility>
#include <vector>

struct LogRecord {
    std::string user;
    std::string action;
};

bool append_limited(std::vector<LogRecord>& records, LogRecord next) {
    constexpr std::size_t max_records = 5000;
    constexpr std::size_t max_field_length = 80;

    if (records.size() >= max_records) {
        return false;
    }
    if (next.user.empty() || next.user.size() > max_field_length ||
        next.action.empty() || next.action.size() > max_field_length) {
        return false;
    }

    records.push_back(std::move(next));
    return true;
}

Limit Nesting Before Recursive Parsing

#include <optional>
#include <string_view>

bool nesting_depth_allowed(std::string_view json_like_text) {
    constexpr int max_depth = 20;
    int depth = 0;

    for (char ch : json_like_text) {
        if (ch == '{' || ch == '[') {
            if (++depth > max_depth) {
                return false;
            }
        } else if (ch == '}' || ch == ']') {
            if (depth == 0) {
                return false;
            }
            --depth;
        }
    }

    return depth == 0;
}
