Allowlist Command-Line and Environment Values

#include <cstdlib>
#include <iostream>
#include <optional>
#include <string>
#include <string_view>

enum class Mode {
    read_only,
    batch
};

std::optional<Mode> parse_mode(std::string_view value) {
    if (value == "read-only") {
        return Mode::read_only;
    }
    if (value == "batch") {
        return Mode::batch;
    }
    return std::nullopt;
}

std::optional<std::string> parse_input_name(std::string_view value) {
    if (value.empty() || value.size() > 64) {
        return std::nullopt;
    }
    if (value.find('/') != std::string_view::npos ||
        value.find('\\') != std::string_view::npos ||
        value.find("..") != std::string_view::npos) {
        return std::nullopt;
    }
    return std::string(value);
}

int main(int argc, char* argv[]) {
    const char* raw_mode = std::getenv("APP_MODE");
    auto mode = parse_mode(raw_mode == nullptr ? "read-only" : raw_mode);
    auto name = argc == 2 ? parse_input_name(argv[1]) : std::nullopt;

    if (!mode || !name) {
        std::cerr << "invalid command-line or environment input\n";
        return 1;
    }

    std::cout << "accepted input set\n";
    return 0;
}

Treat Callback Data as Boundary Data

#include <optional>
#include <string>
#include <string_view>
#include <utility>

struct DeviceReading {
    std::string sensor_id;
    double volts;
};

std::optional<DeviceReading> parse_device_callback(std::string_view sensor_id, double volts) {
    if (sensor_id.empty() || sensor_id.size() > 24) {
        return std::nullopt;
    }
    if (volts < 0.0 || volts > 5.0) {
        return std::nullopt;
    }

    for (char ch : sensor_id) {
        if (!(ch >= 'A' && ch <= 'Z') && !(ch >= '0' && ch <= '9') && ch != '-') {
            return std::nullopt;
        }
    }

    return DeviceReading{std::string(sensor_id), volts};
}
