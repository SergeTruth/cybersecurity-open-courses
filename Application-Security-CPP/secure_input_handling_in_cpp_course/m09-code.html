Parse Raw Text into a Trusted Request Object

#include <algorithm>
#include <charconv>
#include <cctype>
#include <optional>
#include <sstream>
#include <string>
#include <string_view>
#include <system_error>

struct Request {
    std::string name;
    unsigned int count;
};

bool safe_name(std::string_view name) {
    if (name.empty() || name.size() > 31) {
        return false;
    }
    return std::all_of(name.begin(), name.end(), [](char ch) {
        unsigned char value = static_cast<unsigned char>(ch);
        return std::isalnum(value) || ch == '_' || ch == '-';
    });
}

std::optional<unsigned int> parse_count(std::string_view text) {
    unsigned int value = 0;
    auto [ptr, ec] = std::from_chars(text.data(), text.data() + text.size(), value);
    if (ec != std::errc{} || ptr != text.data() + text.size() || value < 1 || value > 100) {
        return std::nullopt;
    }
    return value;
}

std::optional<Request> parse_request_line(const std::string& line) {
    std::istringstream stream(line);
    std::string name;
    std::string count_text;
    std::string trailing;

    if (!(stream >> name >> count_text) || (stream >> trailing)) {
        return std::nullopt;
    }
    if (!safe_name(name)) {
        return std::nullopt;
    }

    auto count = parse_count(count_text);
    if (!count) {
        return std::nullopt;
    }

    return Request{name, *count};
}

Keep Later Code Behind the Validated Type

#include <iostream>
#include <optional>
#include <string>

bool process_request(const Request& request) {
    for (unsigned int i = 0; i < request.count; ++i) {
        std::cout << "processing " << request.name << '\n';
    }
    return true;
}

bool handle_raw_line(const std::string& raw_line) {
    auto request = parse_request_line(raw_line);
    if (!request) {
        return false;
    }

    return process_request(*request);
}
