Create a Validated Value at the Boundary

#include <algorithm>
#include <cctype>
#include <cstddef>
#include <optional>
#include <string>
#include <string_view>

struct LoginName {
    std::string value;
};

bool is_login_name_char(unsigned char ch) {
    return std::isalnum(ch) || ch == '_' || ch == '-';
}

std::optional<LoginName> parse_login_name(std::string_view raw) {
    constexpr std::size_t max_name = 31;

    if (raw.empty() || raw.size() > max_name) {
        return std::nullopt;
    }

    const bool safe = std::all_of(raw.begin(), raw.end(), [](char ch) {
        return is_login_name_char(static_cast<unsigned char>(ch));
    });

    if (!safe) {
        return std::nullopt;
    }

    return LoginName{std::string(raw)};
}

Use Only Verified Fields Downstream

#include <charconv>
#include <cstddef>
#include <optional>
#include <string_view>
#include <system_error>

struct PageRequest {
    std::size_t page;
    std::size_t page_size;
};

std::optional<std::size_t> parse_size(std::string_view text, std::size_t min, std::size_t max) {
    std::size_t value = 0;
    const char* first = text.data();
    const char* last = first + text.size();
    auto [ptr, ec] = std::from_chars(first, last, value);

    if (ec != std::errc{} || ptr != last || value < min || value > max) {
        return std::nullopt;
    }

    return value;
}

std::optional<PageRequest> parse_page_request(std::string_view page, std::string_view page_size) {
    auto parsed_page = parse_size(page, 1, 10000);
    auto parsed_size = parse_size(page_size, 1, 100);

    if (!parsed_page || !parsed_size) {
        return std::nullopt;
    }

    return PageRequest{*parsed_page, *parsed_size};
}
