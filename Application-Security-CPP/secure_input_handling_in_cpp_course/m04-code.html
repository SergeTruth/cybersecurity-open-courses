Read a Line and Enforce the Policy Limit

#include <cstddef>
#include <istream>
#include <optional>
#include <string>

std::optional<std::string> read_required_line(std::istream& input, std::size_t max_length) {
    std::string line;

    if (!std::getline(input, line)) {
        return std::nullopt;
    }
    if (line.empty() || line.size() > max_length) {
        return std::nullopt;
    }
    if (line.find('\0') != std::string::npos) {
        return std::nullopt;
    }

    return line;
}

Reject Partial Parses and Trailing Data

#include <optional>
#include <sstream>
#include <string>

struct UserLimit {
    std::string name;
    int limit;
};

std::optional<UserLimit> parse_user_limit(const std::string& line) {
    std::istringstream stream(line);
    UserLimit parsed{};
    std::string trailing;

    if (!(stream >> parsed.name >> parsed.limit)) {
        return std::nullopt;
    }
    if (stream >> trailing) {
        return std::nullopt;
    }
    if (parsed.name.size() > 31 || parsed.limit < 1 || parsed.limit > 100) {
        return std::nullopt;
    }

    return parsed;
}
