Return a Visible Validation Status

#include <charconv>
#include <optional>
#include <string_view>
#include <system_error>

enum class ParseError {
    none,
    empty,
    malformed,
    out_of_range
};

struct ParseResult {
    std::optional<int> value;
    ParseError error;
};

ParseResult parse_retry_count(std::string_view text) {
    if (text.empty()) {
        return {std::nullopt, ParseError::empty};
    }

    int value = 0;
    auto [ptr, ec] = std::from_chars(text.data(), text.data() + text.size(), value);
    if (ec != std::errc{} || ptr != text.data() + text.size()) {
        return {std::nullopt, ParseError::malformed};
    }
    if (value < 0 || value > 10) {
        return {std::nullopt, ParseError::out_of_range};
    }

    return {value, ParseError::none};
}

Test Boundaries and Malformed Input

#include <cassert>

void test_parse_retry_count() {
    assert(parse_retry_count("").error == ParseError::empty);
    assert(parse_retry_count("x").error == ParseError::malformed);
    assert(parse_retry_count("1x").error == ParseError::malformed);
    assert(parse_retry_count("-1").error == ParseError::out_of_range);
    assert(parse_retry_count("11").error == ParseError::out_of_range);

    auto zero = parse_retry_count("0");
    auto ten = parse_retry_count("10");

    assert(zero.value && *zero.value == 0);
    assert(ten.value && *ten.value == 10);
}
