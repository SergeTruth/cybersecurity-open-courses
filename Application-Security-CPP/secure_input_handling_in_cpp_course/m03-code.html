Copy to a C API Buffer Without Truncation

#include <array>
#include <cstring>
#include <optional>
#include <string_view>

std::optional<std::array<char, 32>> make_c_api_name(std::string_view input) {
    std::array<char, 32> buffer{};

    if (input.empty() || input.size() >= buffer.size()) {
        return std::nullopt;
    }
    if (input.find('\0') != std::string_view::npos) {
        return std::nullopt;
    }

    std::memcpy(buffer.data(), input.data(), input.size());
    buffer[input.size()] = '\0';
    return buffer;
}

Keep string_view Tied to Owned Storage

#include <optional>
#include <string>
#include <string_view>

class ParsedToken {
public:
    static std::optional<ParsedToken> create(std::string raw) {
        if (raw.empty() || raw.size() > 80) {
            return std::nullopt;
        }
        return ParsedToken(std::move(raw));
    }

    std::string_view view() const {
        return storage_;
    }

private:
    explicit ParsedToken(std::string raw) : storage_(std::move(raw)) {}

    std::string storage_;
};
