Create a Version Manifest Before Changes

MANIFEST=gpu-ai-stack-manifest.txt

{
  printf '[timestamp]\n%s\n\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)"
  printf '[os]\n'
  cat /etc/os-release 2>/dev/null || true
  printf '\n[kernel]\n'
  uname -r
  printf '\n[gpu hardware]\n'
  lspci -nn | grep -Ei 'nvidia|amd|ati|vga|3d|display' || true
  printf '\n[nvidia]\n'
  nvidia-smi --query-gpu=name,driver_version,cuda_version --format=csv 2>/dev/null || true
  printf '\n[amd rocm]\n'
  rocminfo 2>/dev/null | grep -E 'Name:|Marketing Name|Runtime Version' | head -40 || true
} | tee "$MANIFEST"

Capture Package Versions Without Installing Anything

printf '[debian packages]\n'
dpkg-query -W 'nvidia*' 'cuda*' 'libcuda*' 'rocm*' 'amdgpu*' 2>/dev/null || true

printf '\n[debian package policy]\n'
apt-cache policy 'nvidia-driver*' 'cuda-toolkit*' 'rocm*' 2>/dev/null || true

printf '\n[rpm packages]\n'
rpm -qa '*nvidia*' '*cuda*' '*rocm*' '*amdgpu*' 2>/dev/null | sort || true

printf '\n[held packages]\n'
apt-mark showhold 2>/dev/null || true

Write a Rebuild Checklist Artifact

from pathlib import Path

steps = [
    "Confirm distribution, kernel, and Secure Boot policy.",
    "Record GPU hardware and current driver stack.",
    "Choose one package source strategy: distribution or vendor.",
    "Confirm kernel headers and DKMS/module rebuild path.",
    "Install or update during a maintenance window.",
    "Validate host GPU tools and AI runtime smoke tests.",
    "Validate container GPU access if containers are used.",
    "Record rollback point, package versions, and validation output.",
]

Path("gpu-driver-rebuild-checklist.md").write_text(
    "\n".join(f"- [ ] {step}" for step in steps),
    encoding="utf-8",
)
