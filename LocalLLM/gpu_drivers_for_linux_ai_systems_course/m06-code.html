Linux GPU Code Examples

Check Host Container Runtime GPU Configuration

printf '[container runtime]\n'
docker info 2>/dev/null | grep -E 'Runtimes|Default Runtime|Cgroup Driver' || true

printf '\n[nvidia container toolkit]\n'
nvidia-container-cli info 2>/dev/null | sed -n '1,80p' || echo "nvidia-container-cli not available"

printf '\n[host device nodes]\n'
ls -l /dev/nvidia* /dev/dri/renderD* /dev/kfd 2>/dev/null || true

Run an NVIDIA Container Visibility Check

IMAGE=nvidia/cuda:12.4.1-base-ubuntu22.04

docker run --rm --gpus all "$IMAGE" nvidia-smi

docker run --rm --gpus all "$IMAGE" sh -lc '
  echo "[devices]"
  ls -l /dev/nvidia* 2>/dev/null || true
  echo "[libraries]"
  ldconfig -p 2>/dev/null | grep -E "libcuda|libnvidia" || true
'

Run a ROCm Container Visibility Check

IMAGE=rocm/rocm-terminal:latest

docker run --rm \
  --device=/dev/kfd \
  --device=/dev/dri \
  --group-add video \
  --group-add render \
  "$IMAGE" rocminfo
