Run a Compact End-to-End Validation Checklist

REPORT=gpu-ai-validation-summary.txt

{
  printf '[kernel]\n'
  uname -r
  printf '\n[hardware]\n'
  lspci -nn | grep -Ei 'nvidia|amd|ati|vga|3d|display' || true
  printf '\n[modules]\n'
  lsmod | grep -Ei 'nvidia|amdgpu|drm' || true
  printf '\n[nvidia status]\n'
  nvidia-smi --query-gpu=name,driver_version,cuda_version,memory.total --format=csv 2>/dev/null || true
  printf '\n[amd status]\n'
  rocminfo 2>/dev/null | grep -E 'Name:|Runtime Version' | head -30 || true
  printf '\n[devices]\n'
  ls -l /dev/nvidia* /dev/dri/renderD* /dev/kfd 2>/dev/null || true
} | tee "$REPORT"

Track Evidence Needed for a Recoverable Stack

from pathlib import Path

required_artifacts = {
    "stack_manifest": "gpu-ai-stack-manifest.txt",
    "validation_summary": "gpu-ai-validation-summary.txt",
    "python_smoke_test": "gpu-python-smoke.json",
    "container_check": "container-gpu-check.txt",
    "package_versions": "gpu-package-versions.txt",
    "rollback_plan": "gpu-driver-rebuild-checklist.md",
}

for name, path in required_artifacts.items():
    status = "present" if Path(path).exists() else "missing"
    print({"artifact": name, "path": path, "status": status})
