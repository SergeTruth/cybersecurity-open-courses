Run a Small Matrix Performance Smoke Test

import time
import torch

if not torch.cuda.is_available():
    raise SystemExit("GPU runtime is not visible; refusing to report a CPU fallback as success")

device = torch.device("cuda:0")
torch.cuda.reset_peak_memory_stats(device)

start = time.perf_counter()
x = torch.randn((4096, 4096), device=device)
for _ in range(10):
    x = x @ x.T
torch.cuda.synchronize()
elapsed = time.perf_counter() - start

print({
    "device": torch.cuda.get_device_name(device),
    "elapsed_seconds": round(elapsed, 3),
    "peak_memory_mb": round(torch.cuda.max_memory_allocated(device) / 1024 / 1024, 1),
})

Validate an Inference Server Uses GPU

SERVER=http://127.0.0.1:8000

curl -s "$SERVER/health" | tee inference-health.json
curl -s "$SERVER/v1/models" | tee inference-models.json

before=$(nvidia-smi --query-gpu=utilization.gpu,memory.used --format=csv,noheader,nounits 2>/dev/null || true)
curl -s "$SERVER/v1/completions" \
  -H 'content-type: application/json' \
  -d '{"model":"local-model","prompt":"Say ready.","max_tokens":8}' | tee inference-smoke.json
after=$(nvidia-smi --query-gpu=utilization.gpu,memory.used --format=csv,noheader,nounits 2>/dev/null || true)

printf 'before=%s\nafter=%s\n' "$before" "$after" | tee inference-gpu-delta.txt
