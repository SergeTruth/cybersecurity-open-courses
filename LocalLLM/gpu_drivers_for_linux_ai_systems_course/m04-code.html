Linux GPU Code Examples

Inspect AMDGPU and ROCm Signals

printf '[amd pci devices]\n'
lspci -nn | grep -Ei 'amd|ati|radeon|vga|3d' || true

printf '\n[amdgpu module]\n'
lsmod | grep amdgpu || true
modinfo amdgpu 2>/dev/null | sed -n '1,16p' || true

printf '\n[rocm tools]\n'
rocminfo 2>/dev/null | sed -n '1,80p' || echo "rocminfo not available"
rocm-smi 2>/dev/null || echo "rocm-smi not available"

Check OpenCL and Vulkan Visibility

printf '[opencl]\n'
clinfo 2>/dev/null | grep -E 'Platform Name|Device Name|Device Version' || echo "clinfo not available"

printf '\n[vulkan]\n'
vulkaninfo --summary 2>/dev/null || echo "vulkaninfo not available"

printf '\n[render nodes]\n'
ls -l /dev/dri/renderD* 2>/dev/null || true

Check PyTorch ROCm Availability

import torch

report = {
    "torch": torch.__version__,
    "hip_build": getattr(torch.version, "hip", None),
    "cuda_api_available": torch.cuda.is_available(),
    "device_count": torch.cuda.device_count(),
}

if torch.cuda.is_available():
    report["device_names"] = [torch.cuda.get_device_name(i) for i in range(torch.cuda.device_count())]

print(report)
