Linux GPU Code Examples

Collect Kernel, DKMS, and Driver Logs

CASE=gpu-troubleshooting-$(date -u +%Y%m%dT%H%M%SZ)
mkdir -p "$CASE"

uname -a | tee "$CASE/kernel.txt"
dkms status 2>/dev/null | tee "$CASE/dkms-status.txt" || true
journalctl -k -b --no-pager 2>/dev/null |
  grep -Ei 'nvidia|amdgpu|drm|firmware|dkms|module|secure boot' |
  tee "$CASE/kernel-gpu-messages.txt" || true
dmesg 2>/dev/null |
  grep -Ei 'nvidia|amdgpu|drm|firmware|dkms|module|secure boot' |
  tee "$CASE/dmesg-gpu-messages.txt" || true

Check Secure Boot and Module Signing Clues

printf '[secure boot]\n'
mokutil --sb-state 2>/dev/null || echo "mokutil not available"

printf '\n[module signers]\n'
for module in nvidia nvidia_uvm amdgpu; do
  printf '\n[%s]\n' "$module"
  modinfo "$module" 2>/dev/null | grep -E '^(filename|version|signer|sig_key|sig_hashalgo):' || true
done

Create a Layered Troubleshooting Report

from pathlib import Path

checks = [
    ("hardware_detected", "lspci output contains expected GPU"),
    ("kernel_module_loaded", "lsmod contains nvidia, amdgpu, or expected driver"),
    ("device_nodes_present", "/dev/nvidia* or /dev/dri/renderD* exists"),
    ("user_permissions", "service user belongs to render/video or vendor-required group"),
    ("runtime_visible", "nvidia-smi, rocminfo, or framework smoke test succeeds"),
    ("container_visible", "container sees devices and runtime libraries"),
    ("no_recent_kernel_breakage", "dkms and kernel logs show clean module load"),
]

lines = ["# GPU Driver Troubleshooting Checklist", ""]
for key, description in checks:
    lines.append(f"- [ ] **{key}**: {description}")

Path("gpu-driver-troubleshooting-checklist.md").write_text("\n".join(lines), encoding="utf-8")
