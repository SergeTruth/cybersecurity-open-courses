Query NVIDIA Driver, GPU, and Process State

nvidia-smi

nvidia-smi --query-gpu=index,name,driver_version,cuda_version,memory.total,memory.used,utilization.gpu,temperature.gpu,power.draw \
  --format=csv,noheader,nounits |
  tee nvidia-gpu-status.csv

nvidia-smi pmon -c 1 2>/dev/null | tee nvidia-process-snapshot.txt || true

Compare Driver, Toolkit, and Library Signals

printf '[driver module]\n'
modinfo nvidia 2>/dev/null | grep -E '^(version|filename|signer):' || true

printf '\n[nvidia-smi cuda runtime signal]\n'
nvidia-smi --query-gpu=driver_version,cuda_version --format=csv,noheader 2>/dev/null || true

printf '\n[nvcc toolkit signal]\n'
nvcc --version 2>/dev/null || echo "CUDA toolkit compiler not found in PATH"

printf '\n[user-space libraries]\n'
ldconfig -p 2>/dev/null | grep -E 'libcuda|libcudart|libcudnn' || true

Run a Minimal PyTorch CUDA Smoke Test

import time
import torch

if not torch.cuda.is_available():
    raise SystemExit("CUDA is not available to this Python environment")

device = torch.device("cuda:0")
name = torch.cuda.get_device_name(device)

start = time.perf_counter()
x = torch.randn((2048, 2048), device=device)
y = x @ x.T
torch.cuda.synchronize()
elapsed = time.perf_counter() - start

print({
    "device": name,
    "torch": torch.__version__,
    "cuda_build": torch.version.cuda,
    "elapsed_seconds": round(elapsed, 4),
    "allocated_mb": round(torch.cuda.memory_allocated(device) / 1024 / 1024, 1),
    "result_shape": tuple(y.shape),
})
