Review Kernel Modules, Headers, and DKMS Status

KERNEL=$(uname -r)

printf '[running kernel]\n%s\n\n' "$KERNEL"
printf '[kernel headers]\n'
test -e "/lib/modules/$KERNEL/build" && echo "headers present" || echo "headers missing"

printf '\n[dkms status]\n'
dkms status 2>/dev/null || echo "dkms not installed or no dkms modules"

printf '\n[gpu module details]\n'
for module in nvidia amdgpu i915; do
  modinfo "$module" 2>/dev/null | sed -n '1,12p' || true
done

Check Device Nodes and Access Groups

printf '[current user]\n'
id

printf '\n[gpu device nodes]\n'
ls -l /dev/nvidia* /dev/dri/renderD* /dev/kfd 2>/dev/null || true

printf '\n[relevant groups]\n'
getent group video render 2>/dev/null || true

printf '\n[user-space gpu libraries]\n'
ldconfig -p 2>/dev/null | grep -Ei 'cuda|nvidia|rocm|hip|vulkan|opencl' | head -80 || true

Summarize Stack Layers as JSON

import json
import subprocess
from pathlib import Path

def run(command):
    result = subprocess.run(command, text=True, capture_output=True, check=False)
    return result.stdout.strip()

report = {
    "kernel": run(["uname", "-r"]),
    "gpu_pci": run(["sh", "-c", "lspci -nn | grep -Ei 'vga|3d|display|nvidia|amd|ati|intel' || true"]).splitlines(),
    "device_nodes": [str(path) for path in Path("/dev").glob("nvidia*")] + [str(path) for path in Path("/dev/dri").glob("renderD*")],
    "loaded_modules": run(["sh", "-c", "lsmod | grep -Ei 'nvidia|amdgpu|i915|drm' || true"]).splitlines(),
}

print(json.dumps(report, indent=2))
