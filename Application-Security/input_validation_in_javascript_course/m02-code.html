function parsePositiveInteger(value, name, maximum) {
  const number = Number(value);
  if (!Number.isInteger(number) || number < 1 || number > maximum) {
    throw new Error(`${name} must be an integer from 1 to ${maximum}`);
  }
  return number;
}

function parseShortText(value, name, maximum) {
  if (typeof value !== "string") {
    throw new Error(`${name} must be text`);
  }

  const text = value.trim();
  if (text.length === 0 || text.length > maximum) {
    throw new Error(`${name} must be 1 to ${maximum} characters`);
  }
  return text;
}

const query = new URLSearchParams("?pageSize=25");
const storageValue = JSON.parse('{"displayName":"Ada"}');

const pageSize = parsePositiveInteger(query.get("pageSize"), "pageSize", 100);
const displayName = parseShortText(storageValue.displayName, "displayName", 80);

console.log({ pageSize, displayName });
