const ALLOWED_STATUSES = new Set(["draft", "submitted", "approved"]);

function parseOrder(raw) {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) {
    throw new Error("order must be an object");
  }

  const sku = raw.sku;
  const quantity = raw.quantity;
  const status = raw.status;

  if (typeof sku !== "string" || !/^[A-Z0-9-]{3,20}$/.test(sku)) {
    throw new Error("sku must be 3 to 20 uppercase letters, numbers, or dashes");
  }

  if (!Number.isInteger(quantity) || quantity < 1 || quantity > 100) {
    throw new Error("quantity must be between 1 and 100");
  }

  if (!ALLOWED_STATUSES.has(status)) {
    throw new Error("status is not allowed");
  }

  return { sku, quantity, status };
}

console.log(parseOrder({ sku: "ABC-123", quantity: 2, status: "draft" }));
