const ALLOWED_PROFILE_FIELDS = new Set(["displayName", "timezone"]);
const BLOCKED_KEYS = new Set(["__proto__", "prototype", "constructor"]);

function safeProfileUpdate(raw) {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) {
    throw new Error("payload must be an object");
  }

  for (const key of Object.keys(raw)) {
    if (BLOCKED_KEYS.has(key) || !ALLOWED_PROFILE_FIELDS.has(key)) {
      throw new Error(`field is not allowed: ${key}`);
    }
  }

  const displayName = String(raw.displayName ?? "").trim();
  const timezone = String(raw.timezone ?? "").trim();

  if (displayName.length < 1 || displayName.length > 60) {
    throw new Error("displayName is invalid");
  }
  if (!/^[A-Za-z_]+\/[A-Za-z_]+$/.test(timezone)) {
    throw new Error("timezone is invalid");
  }

  return Object.freeze({ displayName, timezone });
}

console.log(safeProfileUpdate({ displayName: "Grace", timezone: "America/New_York" }));
