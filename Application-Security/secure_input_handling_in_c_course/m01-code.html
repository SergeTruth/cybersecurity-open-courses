#include <ctype.h>
#include <stdio.h>
#include <string.h>

#define USERNAME_CAP 32

static int bounded_length(const char *text, size_t limit, size_t *length) {
    for (size_t i = 0; i < limit; i++) {
        if (text[i] == '\0') {
            *length = i;
            return 0;
        }
    }
    return -1;
}

static int valid_username_char(unsigned char ch) {
    return isalnum(ch) || ch == '_' || ch == '-';
}

int copy_valid_username(char dest[USERNAME_CAP], const char *input) {
    size_t length;

    if (input == NULL) {
        return -1;
    }

    if (bounded_length(input, USERNAME_CAP, &length) != 0 || length == 0) {
        return -1;
    }

    for (size_t i = 0; i < length; i++) {
        if (!valid_username_char((unsigned char)input[i])) {
            return -1;
        }
    }

    memcpy(dest, input, length + 1);
    return 0;
}

int main(int argc, char **argv) {
    char username[USERNAME_CAP];

    if (argc != 2 || copy_valid_username(username, argv[1]) != 0) {
        fputs("usage: program <1-31 safe username>\n", stderr);
        return 1;
    }

    printf("accepted username: %s\n", username);
    return 0;
}
