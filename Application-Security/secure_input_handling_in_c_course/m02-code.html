#include <stdio.h>
#include <stdlib.h>
#include <string.h>

enum app_mode {
    MODE_READ_ONLY,
    MODE_BATCH
};

int parse_mode_from_env(enum app_mode *mode) {
    const char *value = getenv("APP_MODE");

    if (value == NULL || strcmp(value, "read-only") == 0) {
        *mode = MODE_READ_ONLY;
        return 0;
    }
    if (strcmp(value, "batch") == 0) {
        *mode = MODE_BATCH;
        return 0;
    }

    return -1;
}

int parse_input_name(const char *arg) {
    if (arg == NULL || arg[0] == '\0') {
        return -1;
    }
    if (strchr(arg, '/') != NULL || strstr(arg, "..") != NULL) {
        return -1;
    }
    if (strlen(arg) > 64) {
        return -1;
    }
    return 0;
}

int main(int argc, char **argv) {
    enum app_mode mode;

    if (argc != 2 || parse_mode_from_env(&mode) != 0 || parse_input_name(argv[1]) != 0) {
        fputs("invalid command-line or environment input\n", stderr);
        return 1;
    }

    puts(mode == MODE_BATCH ? "batch mode" : "read-only mode");
    return 0;
}
