#include <ctype.h>
#include <stdio.h>
#include <string.h>

#define BASE_DIR "uploads"
#define PATH_CAP 128
#define MAX_FILE_BYTES 1048576L

static int safe_file_name(const char *name) {
    size_t length;

    if (name == NULL) {
        return 0;
    }
    length = strlen(name);
    if (length == 0 || length > 64) {
        return 0;
    }
    if (strstr(name, "..") != NULL || strchr(name, '/') != NULL || strchr(name, '\\') != NULL) {
        return 0;
    }
    if (length < 4 || strcmp(name + length - 4, ".cfg") != 0) {
        return 0;
    }

    for (size_t i = 0; i < length; i++) {
        unsigned char ch = (unsigned char)name[i];
        if (!isalnum(ch) && ch != '_' && ch != '-' && ch != '.') {
            return 0;
        }
    }
    return 1;
}

int open_valid_config(const char *name, FILE **out) {
    char path[PATH_CAP];
    int written;
    FILE *file;

    if (!safe_file_name(name)) {
        return -1;
    }

    written = snprintf(path, sizeof(path), "%s/%s", BASE_DIR, name);
    if (written < 0 || (size_t)written >= sizeof(path)) {
        return -1;
    }

    file = fopen(path, "rb");
    if (file == NULL) {
        return -1;
    }
    if (fseek(file, 0, SEEK_END) != 0 || ftell(file) > MAX_FILE_BYTES) {
        fclose(file);
        return -1;
    }
    rewind(file);

    *out = file;
    return 0;
}
