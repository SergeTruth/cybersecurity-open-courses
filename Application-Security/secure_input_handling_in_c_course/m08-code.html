#include <assert.h>
#include <errno.h>
#include <stdlib.h>

int parse_count(const char *text, int *count) {
    char *end = NULL;
    long value;

    if (text == NULL || text[0] == '\0') {
        return -1;
    }

    errno = 0;
    value = strtol(text, &end, 10);
    if (errno != 0 || end == text || *end != '\0') {
        return -1;
    }
    if (value < 1 || value > 100) {
        return -1;
    }

    *count = (int)value;
    return 0;
}

static void test_parse_count(void) {
    int value = 0;

    assert(parse_count("1", &value) == 0 && value == 1);
    assert(parse_count("100", &value) == 0 && value == 100);

    assert(parse_count("0", &value) != 0);
    assert(parse_count("101", &value) != 0);
    assert(parse_count("-1", &value) != 0);
    assert(parse_count("12x", &value) != 0);
    assert(parse_count("", &value) != 0);
}

int main(void) {
    test_parse_count();
    return 0;
}
