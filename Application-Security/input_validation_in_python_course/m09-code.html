ALLOWED_STATUSES = {"draft", "submitted", "approved", "rejected"}
ALLOWED_TRANSITIONS = {
    "draft": {"submitted"},
    "submitted": {"approved", "rejected"},
    "approved": set(),
    "rejected": set(),
}


class ValidationError(ValueError):
    pass


def validate_ticket_update(current_status: str, payload: dict) -> dict:
    unexpected = set(payload) - {"title", "status"}
    if unexpected:
        raise ValidationError(f"unexpected fields: {sorted(unexpected)}")

    title = payload.get("title")
    status = payload.get("status")

    if not isinstance(title, str) or not 1 <= len(title.strip()) <= 120:
        raise ValidationError("title must be 1 to 120 characters")

    if status not in ALLOWED_STATUSES:
        raise ValidationError("status is not allowed")

    if status not in ALLOWED_TRANSITIONS.get(current_status, set()):
        raise ValidationError("status transition is not allowed")

    return {"title": title.strip(), "status": status}
