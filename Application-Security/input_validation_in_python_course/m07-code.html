import sqlite3
import subprocess


ALLOWED_PROFILE_FIELDS = {"display_name", "timezone"}
ALLOWED_REPORTS = {"daily", "weekly", "monthly"}


class ValidationError(ValueError):
    pass


def update_profile(conn: sqlite3.Connection, user_id: int, changes: dict) -> None:
    unexpected = set(changes) - ALLOWED_PROFILE_FIELDS
    if unexpected:
        raise ValidationError(f"fields are not allowed: {sorted(unexpected)}")

    display_name = str(changes.get("display_name", "")).strip()
    timezone = str(changes.get("timezone", "")).strip()
    if not display_name or len(display_name) > 60:
        raise ValidationError("display_name is invalid")
    if "/" not in timezone or len(timezone) > 64:
        raise ValidationError("timezone is invalid")

    conn.execute(
        "UPDATE profiles SET display_name = ?, timezone = ? WHERE user_id = ?",
        (display_name, timezone, user_id),
    )


def run_report(report_name: str) -> None:
    if report_name not in ALLOWED_REPORTS:
        raise ValidationError("report is not allowed")

    subprocess.run(["python", "-m", "reports", report_name], check=True)
