from dataclasses import dataclass
from decimal import Decimal, InvalidOperation


class ValidationError(ValueError):
    pass


@dataclass(frozen=True)
class OrderItem:
    sku: str
    quantity: int
    price: Decimal


def parse_order_item(raw: dict) -> OrderItem:
    sku = raw.get("sku")
    quantity = raw.get("quantity")
    price = raw.get("price")

    if not isinstance(sku, str) or not sku.isalnum() or len(sku) > 20:
        raise ValidationError("sku must be alphanumeric and 20 characters or fewer")

    if not isinstance(quantity, int) or not 1 <= quantity <= 100:
        raise ValidationError("quantity must be between 1 and 100")

    try:
        parsed_price = Decimal(str(price))
    except (InvalidOperation, ValueError):
        raise ValidationError("price must be a decimal number") from None

    if parsed_price <= Decimal("0") or parsed_price > Decimal("9999.99"):
        raise ValidationError("price is outside the allowed range")

    return OrderItem(sku=sku, quantity=quantity, price=parsed_price)


item = parse_order_item({"sku": "ABC123", "quantity": 2, "price": "19.95"})
print(item)
