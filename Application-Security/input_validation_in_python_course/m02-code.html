import json
import os


class ValidationError(ValueError):
    pass


def positive_int(value: str, *, name: str, maximum: int) -> int:
    try:
        number = int(value)
    except (TypeError, ValueError):
        raise ValidationError(f"{name} must be an integer") from None

    if not 1 <= number <= maximum:
        raise ValidationError(f"{name} must be between 1 and {maximum}")
    return number


def short_text(value: object, *, name: str, maximum: int) -> str:
    if not isinstance(value, str):
        raise ValidationError(f"{name} must be text")

    text = value.strip()
    if not text or len(text) > maximum:
        raise ValidationError(f"{name} must be 1 to {maximum} characters")
    return text


page_size = positive_int(os.getenv("PAGE_SIZE", "25"), name="PAGE_SIZE", maximum=100)
payload = json.loads('{"username": "alice", "display_name": "Alice"}')

username = short_text(payload.get("username"), name="username", maximum=40)
display_name = short_text(payload.get("display_name"), name="display_name", maximum=80)

print(page_size, username, display_name)
