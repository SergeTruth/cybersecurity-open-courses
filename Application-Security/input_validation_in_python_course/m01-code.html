from dataclasses import dataclass


class ValidationError(ValueError):
    pass


@dataclass(frozen=True)
class Signup:
    email: str
    age: int


def parse_signup(raw: dict) -> Signup:
    email = raw.get("email")
    age = raw.get("age")

    if not isinstance(email, str):
        raise ValidationError("email is required")

    email = email.strip().lower()
    if "@" not in email or len(email) > 254:
        raise ValidationError("email is invalid")

    if not isinstance(age, int) or not 13 <= age <= 120:
        raise ValidationError("age must be between 13 and 120")

    return Signup(email=email, age=age)


signup = parse_signup({"email": "User@example.com ", "age": 30})
print(signup)
