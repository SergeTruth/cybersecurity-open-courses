import json
from pathlib import Path


class ValidationError(ValueError):
    pass


BASE_DIR = Path("/srv/app/uploads").resolve()
MAX_BYTES = 1_000_000


def resolve_upload_path(filename: str) -> Path:
    if not isinstance(filename, str) or not filename:
        raise ValidationError("filename is required")

    candidate = (BASE_DIR / filename).resolve()
    try:
        candidate.relative_to(BASE_DIR)
    except ValueError:
        raise ValidationError("filename escapes the upload directory") from None

    if candidate.suffix.lower() != ".json":
        raise ValidationError("only JSON files are accepted")
    return candidate


def load_customer_import(filename: str) -> list[dict]:
    path = resolve_upload_path(filename)
    if path.stat().st_size > MAX_BYTES:
        raise ValidationError("file is too large")

    data = json.loads(path.read_text(encoding="utf-8"))
    customers = data.get("customers") if isinstance(data, dict) else None
    if not isinstance(customers, list):
        raise ValidationError("customers must be a list")

    return customers
